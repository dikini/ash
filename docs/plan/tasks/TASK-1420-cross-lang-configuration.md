## Specifications

### Configuration Schema

```yaml
# cross_lang_config.yaml
version: 1
rust_crates:
  - name: "ash-core"
    modules: ["effect", "capability", "workflow"]
    exclude: ["tests/*", "benches/*"]
  - name: "ash-interp"
    modules: ["runtime", "execute", "capability_provenance"]
    exclude: ["tests/*"]

ash_extensions:
  - ".ash"

mappings:
  # Hardcoded high-value mappings
  - ash_symbol: "Effect"
    ash_kind: "type"
    rust_symbol: "ash_core::effect::Effect"
    rust_kind: "enum"
    confidence: "high"

  # Auto-generated mappings (built by build script)
  - ash_symbol: "Capability"
    ash_kind: "type"
    rust_symbol: "ash_core::capability::Capability"
    rust_kind: "trait"
    confidence: "high"
    source: "auto"
```

### Rust Data Structures

```rust
#[derive(Debug, Deserialize, Serialize)]
pub struct CrossLangConfig {
    pub version: u32,
    pub rust_crates: Vec<RustCrate>,
    pub ash_extensions: Vec<String>,
    pub mappings: Vec<SymbolMapping>,
}

#[derive(Debug, Deserialize, Serialize)]
pub struct SymbolMapping {
    pub ash_symbol: String,
    pub ash_kind: String,
    pub rust_symbol: String,
    pub rust_kind: String,
    pub confidence: ConfidenceLevel,
    #[serde(default)]
    pub source: MappingSource,
}

#[derive(Debug, Deserialize, Serialize, PartialEq)]
pub enum ConfidenceLevel {
    High,   // Manually verified, stable API
    Medium, // Auto-generated but stable
    Low,    // Heuristic, may change
}

#[derive(Debug, Deserialize, Serialize, PartialEq, Default)]
pub enum MappingSource {
    #[default]
    Manual,
    AutoGenerated,
    Heuristic,
}
```

## Testing Strategy

```rust
#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_load_valid_config() {
        let config = CrossLangConfig::from_file("tests/fixtures/valid_config.yaml").unwrap();
        assert_eq!(config.version, 1);
        assert!(!config.mappings.is_empty());
    }

    #[test]
    fn test_reject_invalid_crate_names() {
        let invalid_config = r#"
        version: 1
        rust_crates:
          - name: "nonexistent-crate"
        "#;
        let result = CrossLangConfig::from_str(invalid_config);
        assert!(result.is_err());
    }

    #[proptest]
    fn test_config_roundtrip(
        version: u32,
        mappings: Vec<SymbolMapping>,
    ) {
        // Test that serialization/deserialization preserves data
        let config = CrossLangConfig { version, mappings, rust_crates: vec![], ash_extensions: vec![] };
        let yaml = serde_yaml::to_string(&config).unwrap();
        let deserialized: CrossLangConfig = serde_yaml::from_str(&yaml).unwrap();
        assert_eq!(config, deserialized);
    }
}
```
